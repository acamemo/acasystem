const el = document.querySelector('input[type="range"]')
const ball = document.querySelector('.ball')
const ballContainer = document.querySelector('.ball-container')

function updateProgressDataAttribute(eventEL) {
  el.dataset.progress = eventEL.value;

  // Using the step increment didn't work for some reason. With the -1 offest, we never make it all
  //  the way to 100%
  const stepIncrement =( getNumberAttribute(eventEL, 'min')/(getNumberAttribute(eventEL, 'max'))) * 100;
  // console.log(stepIncrement, eventEL.value)
  ballContainer.style.transform = `translateX(${(eventEL.value - 1) * (stepIncrement + 1)}%)`
}

function getNumberAttribute(el, attributeName) {
  if (!el || !attributeName) return
  return parseInt(el.getAttribute(attributeName), 10)
}

function setInitialProgress() {
  const el = document.querySelector('input[type="range"]')
  el.dataset.progress = Math.floor((getNumberAttribute(el, 'max') - getNumberAttribute(el, 'min')) / 2);
}


setInitialProgress()
